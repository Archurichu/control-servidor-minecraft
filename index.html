import re, json, time, logging from datetime import datetime from azure.storage.blob import BlobServiceClient # ─── CONFIGURA ESTOS VALORES ───────────────────────────────────────────────── CONNECTION_STRING = "TU_CONNECTION_STRING_AQUI" # lo obtienes en el Paso 2 CONTAINER_NAME = "minecraft-stats" LOG_PATH = "/home/archurichu/bedrock/bedrock_server.log" # ───────────────────────────────────────────────────────────────────────────── logging.basicConfig(level=logging.INFO, format="%(asctime)s %(message)s") jugadores = {} # gamertag → {muertes, primera_vez, ultima_vez} logs_recent = [] # últimas 50 líneas relevantes RE_CONNECT = re.compile(r"Player connected: (.+?),") RE_DISCONNECT = re.compile(r"Player disconnected: (.+?),") RE_DEATH = re.compile(r"(.+?) was (killed|slain|blown|fell|drowned|burned|shot|squashed|pummeled|impaled|skewered|stung|frozen|struck)") def parsear_log(): global logs_recent try: with open(LOG_PATH, "r", errors="ignore") as f: lineas = f.readlines() except FileNotFoundError: return conectados_ahora = set() nuevos_logs = [] for linea in lineas: linea = linea.strip() m = RE_CONNECT.search(linea) if m: nombre = m.group(1).strip() if nombre not in jugadores: jugadores[nombre] = {"muertes": 0, "primera_vez": linea[:19], "ultima_vez": linea[:19]} else: jugadores[nombre]["ultima_vez"] = linea[:19] conectados_ahora.add(nombre) nuevos_logs.append(f"🟢 {nombre} entró al servidor") m = RE_DISCONNECT.search(linea) if m: nombre = m.group(1).strip() conectados_ahora.discard(nombre) nuevos_logs.append(f"🔴 {nombre} salió del servidor") m = RE_DEATH.search(linea) if m: nombre = m.group(1).strip() if nombre in jugadores: jugadores[nombre]["muertes"] += 1 nuevos_logs.append(f"💀 {linea}") logs_recent = nuevos_logs[-50:] stats = { "actualizado": datetime.utcnow().strftime("%Y-%m-%d %H:%M:%S UTC"), "conectados_ahora": list(conectados_ahora), "jugadores": jugadores, "logs": logs_recent } subir_blob(stats) def subir_blob(stats): try: cliente = BlobServiceClient.from_connection_string(CONNECTION_STRING) blob = cliente.get_blob_client(container=CONTAINER_NAME, blob="stats.json") blob.upload_blob( json.dumps(stats, ensure_ascii=False, indent=2), overwrite=True, content_settings={"content_type": "application/json"} ) logging.info("stats.json subido correctamente") except Exception as e: logging.error(f"Error subiendo blob: {e}") if __name__ == "__main__": logging.info("Iniciando recolector de stats...") while True: parsear_log() time.sleep(30)
